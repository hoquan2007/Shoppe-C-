cmake_minimum_required(VERSION 3.24)
project(Shoppe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SHOPPE_POSTGRES "Build PostgreSQL production adapter" OFF)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
if(SHOPPE_POSTGRES)
  find_package(PostgreSQL REQUIRED)
endif()
set(DEPS "${CMAKE_SOURCE_DIR}/.cache/deps")
file(MAKE_DIRECTORY "${DEPS}")
if(NOT EXISTS "${DEPS}/httplib.h")
  file(DOWNLOAD "https://raw.githubusercontent.com/yhirose/cpp-httplib/v0.61.0/httplib.h" "${DEPS}/httplib.h" STATUS DL TLS_VERIFY ON EXPECTED_HASH SHA256=bc77114769db3b3c97e4a6d753dab797c3cadaaa48beead3806298b09702df18)
  if(NOT DL MATCHES "^0;")
    message(FATAL_ERROR "Cannot download cpp-httplib: ${DL}")
  endif()
endif()
if(NOT EXISTS "${DEPS}/json.hpp")
  file(DOWNLOAD "https://raw.githubusercontent.com/nlohmann/json/v3.12.0/single_include/nlohmann/json.hpp" "${DEPS}/json.hpp" STATUS DL TLS_VERIFY ON EXPECTED_HASH SHA256=aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63)
  if(NOT DL MATCHES "^0;")
    message(FATAL_ERROR "Cannot download nlohmann/json: ${DL}")
  endif()
endif()
file(SHA256 "${DEPS}/httplib.h" HTTPLIB_SHA)
file(SHA256 "${DEPS}/json.hpp" JSON_SHA)
if(NOT HTTPLIB_SHA STREQUAL "bc77114769db3b3c97e4a6d753dab797c3cadaaa48beead3806298b09702df18" OR
   NOT JSON_SHA STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
  message(FATAL_ERROR "Cached dependency checksum mismatch; restore the pinned vendor headers.")
endif()
add_library(shoppe_core backend/src/domain.cpp backend/src/database.cpp backend/src/security.cpp backend/src/payments.cpp backend/src/catalog.cpp backend/src/application.cpp)
target_include_directories(shoppe_core PUBLIC backend/include "${DEPS}")
target_compile_definitions(shoppe_core PUBLIC CPPHTTPLIB_OPENSSL_SUPPORT)
if(TARGET SQLite3::SQLite3)
  set(SHOPPE_SQLITE_TARGET SQLite3::SQLite3)
else()
  set(SHOPPE_SQLITE_TARGET SQLite::SQLite3)
endif()
target_link_libraries(shoppe_core PUBLIC OpenSSL::SSL OpenSSL::Crypto ${SHOPPE_SQLITE_TARGET} Threads::Threads)
if(WIN32)
  target_link_libraries(shoppe_core PUBLIC ws2_32 crypt32)
endif()
if(SHOPPE_POSTGRES)
  target_compile_definitions(shoppe_core PUBLIC SHOPPE_POSTGRES)
  target_link_libraries(shoppe_core PUBLIC PostgreSQL::PostgreSQL)
endif()
if(MSVC)
  target_compile_options(shoppe_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(shoppe_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(shoppe backend/src/main.cpp)
target_link_libraries(shoppe PRIVATE shoppe_core)
include(CTest)
if(BUILD_TESTING)
  if(SHOPPE_POSTGRES)
    add_executable(postgres_pool_tests tests/postgres_pool_tests.cpp)
    target_link_libraries(postgres_pool_tests PRIVATE shoppe_core)
  endif()
  add_executable(domain_tests tests/domain_tests.cpp)
  target_link_libraries(domain_tests PRIVATE shoppe_core)
  add_test(NAME domain_tests COMMAND domain_tests)
  add_executable(payment_tests tests/payment_tests.cpp)
  target_link_libraries(payment_tests PRIVATE shoppe_core)
  add_test(NAME payment_tests COMMAND payment_tests)
  set_tests_properties(payment_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_executable(database_tests tests/database_tests.cpp)
  target_link_libraries(database_tests PRIVATE shoppe_core)
  add_test(NAME database_tests COMMAND database_tests)
  set_tests_properties(database_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
endif()
